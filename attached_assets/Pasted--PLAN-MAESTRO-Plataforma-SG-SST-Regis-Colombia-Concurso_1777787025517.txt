## 1. ARQUITECTURA TÉCNICA

```
┌─────────────────────────────────────────────────────────────────────┐
│                        CAPA DE PRESENTACIÓN                         │
│   Replit (React + Vite)                                             │
│   - Dashboard Admin (vista Regis)                                   │
│   - Portal Cliente (vista empresa)                                  │
│   - Módulo Consultor (generación de documentos)                     │
└───────────────────────────┬─────────────────────────────────────────┘
                            │ REST API / Webhooks
┌───────────────────────────▼─────────────────────────────────────────┐
│                        CAPA DE ORQUESTACIÓN                         │
│   n8n (n8n.disier.net)                                              │
│   6 workflows principales + 1 workflow maestro de orquestación      │
│   Claude API como motor de IA en cada workflow                      │
└──────┬──────────────┬────────────────┬──────────────────────────────┘
       │              │                │
┌──────▼──────┐ ┌─────▼──────┐ ┌──────▼──────────┐
│  Supabase   │ │Google Drive│ │  Claude API      │
│  (DB)       │ │(Docs)      │ │  (Skills)        │
│  - empresas │ │  /Regis    │ │  - EXTRACTOR_PDF │
│  - trabajad.│ │  /[empresa]│ │  - MATRIZ_GTC45  │
│  - comités  │ │    /PILA   │ │  - ACTAS         │
│  - cumpli.  │ │    /Examen.│ │  - EMERGENCIAS   │
│  - CIIU     │ │    /Actas  │ │  - CONSULTOR_SST │
└─────────────┘ └────────────┘ └──────────────────┘
       │
┌──────▼──────────────────────────────────────────────────────────────┐
│                    CANALES DE COMUNICACIÓN                          │
│   Gmail API (solicitudes PILA, notif. exámenes)                    │
│   WhatsApp Business API / Wati (seguimiento)                        │
│   Email working → confirmación recibido exámenes médicos           │
└─────────────────────────────────────────────────────────────────────┘
```

**Stack definitivo:**
- **Frontend**: React + Vite + TailwindCSS → Replit (deploy via Replit Deployments)
- **Backend API**: Node.js/Express → mismo Replit (o Replit separado)
- **Automatización**: n8n Cloud (ya conectado vía MCP)
- **Base de datos**: Supabase (ya conectado vía MCP)
- **Storage de documentos**: Google Drive (ya conectado vía MCP)
- **IA core**: Claude API (claude-sonnet-4-20250514)
- **Email**: Gmail API (ya conectado vía MCP)

## 2. PRIORIZACIÓN PARA DEMO 4 DE MAYO

Para mañana, los módulos más demostrables y de mayor impacto visual son:

## 3. ESQUEMA DE BASE DE DATOS — SUPABASE

### Tablas principales

```sql
-- EMPRESAS CLIENTE
CREATE TABLE empresas (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  nombre TEXT NOT NULL,
  nit TEXT UNIQUE NOT NULL,
  codigo_ciiu TEXT NOT NULL,
  descripcion_actividad TEXT,
  tamano TEXT CHECK (tamano IN ('micro','pequeña','mediana','grande')),
  num_empleados INTEGER,
  direccion TEXT,
  ciudad TEXT,
  contacto_nombre TEXT,
  contacto_email TEXT,
  contacto_whatsapp TEXT,
  drive_folder_id TEXT,         -- ID carpeta Google Drive de esta empresa
  nivel_sgsst TEXT CHECK (nivel_sgsst IN ('basico','estandar','avanzado')),
  fecha_inicio_contrato DATE,
  activa BOOLEAN DEFAULT true,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now()
);

-- TRABAJADORES
CREATE TABLE trabajadores (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  empresa_id UUID REFERENCES empresas(id),
  nombres TEXT NOT NULL,
  apellidos TEXT NOT NULL,
  cedula TEXT NOT NULL,
  cargo TEXT,
  area TEXT,
  email TEXT,
  telefono TEXT,
  tipo_contrato TEXT,
  fecha_ingreso DATE,
  activo BOOLEAN DEFAULT true,
  created_at TIMESTAMPTZ DEFAULT now()
);

-- PLANTILLA PILA (mensual por empresa)
CREATE TABLE registros_pila (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  empresa_id UUID REFERENCES empresas(id),
  periodo TEXT NOT NULL,         -- formato: 2026-04
  estado TEXT CHECK (estado IN ('pendiente','solicitado','recibido','archivado')) DEFAULT 'pendiente',
  fecha_solicitud TIMESTAMPTZ,
  fecha_recepcion TIMESTAMPTZ,
  drive_file_id TEXT,            -- ID archivo en Drive
  intentos_solicitud INTEGER DEFAULT 0,
  canal_ultimo_contacto TEXT,   -- email, whatsapp
  observaciones TEXT,
  created_at TIMESTAMPTZ DEFAULT now()
);

-- EXÁMENES MÉDICOS
CREATE TABLE examenes_medicos (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  trabajador_id UUID REFERENCES trabajadores(id),
  empresa_id UUID REFERENCES empresas(id),
  tipo TEXT CHECK (tipo IN ('ingreso','periodico','egreso','post_incapacidad')),
  fecha_examen DATE,
  medico TEXT,
  concepto TEXT CHECK (concepto IN ('apto','apto_con_restricciones','no_apto','pendiente')),
  restricciones JSONB,           -- array de restricciones extraídas por IA
  recomendaciones JSONB,         -- array de recomendaciones
  texto_completo TEXT,           -- transcripción completa del PDF
  drive_file_id TEXT,
  email_confirmacion_enviado BOOLEAN DEFAULT false,
  email_confirmacion_recibido BOOLEAN DEFAULT false,
  created_at TIMESTAMPTZ DEFAULT now()
);

-- COMITÉS (COPASST y Convivencia)
CREATE TABLE comites (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  empresa_id UUID REFERENCES empresas(id),
  tipo TEXT CHECK (tipo IN ('COPASST','convivencia')),
  vigencia_inicio DATE,
  vigencia_fin DATE,            -- 2 años según normativa
  activo BOOLEAN DEFAULT true,
  created_at TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE miembros_comite (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  comite_id UUID REFERENCES comites(id),
  trabajador_id UUID REFERENCES trabajadores(id),
  rol TEXT CHECK (rol IN ('presidente','secretario','vocal_titular','vocal_suplente','representante_empleador','representante_trabajadores')),
  tipo TEXT CHECK (tipo IN ('titular','suplente')),
  vigencia_inicio DATE,
  vigencia_fin DATE,
  activo BOOLEAN DEFAULT true,
  -- LOG de cambios
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now(),
  updated_by TEXT                -- email del consultor que hizo el cambio
);

-- Historial de cambios en comités
CREATE TABLE miembros_comite_log (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  miembro_id UUID REFERENCES miembros_comite(id),
  accion TEXT,                  -- 'creado', 'modificado', 'desactivado'
  campo_modificado TEXT,
  valor_anterior TEXT,
  valor_nuevo TEXT,
  modificado_por TEXT,
  modificado_en TIMESTAMPTZ DEFAULT now()
);

-- ACTAS DE COMITÉ
CREATE TABLE actas_comite (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  comite_id UUID REFERENCES comites(id),
  empresa_id UUID REFERENCES empresas(id),
  numero_acta TEXT,             -- Ej: COPASST-2026-04
  fecha_reunion TIMESTAMPTZ,
  lugar TEXT,
  transcripcion_audio TEXT,    -- transcripción automática de la reunión
  puntos_tratados JSONB,       -- array ingresado por consultor
  acta_generada TEXT,          -- texto completo del acta generada por IA
  drive_file_id TEXT,          -- Word/PDF generado
  estado TEXT CHECK (estado IN ('borrador','revisado','firmado','archivado')) DEFAULT 'borrador',
  asistentes_confirmados JSONB, -- array de IDs con timestamps de confirmación
  link_asistencia TEXT,        -- link único para registro digital de asistencia
  created_at TIMESTAMPTZ DEFAULT now()
);

-- MATRICES DE RIESGO GTC-45
CREATE TABLE matrices_riesgo (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  empresa_id UUID REFERENCES empresas(id),
  version INTEGER DEFAULT 1,
  codigo_ciiu TEXT,
  contenido_json JSONB,         -- estructura completa de la matriz
  estado TEXT CHECK (estado IN ('borrador','revisado','aprobado_arl')) DEFAULT 'borrador',
  drive_file_id_borrador TEXT,
  drive_file_id_arl TEXT,       -- versión aprobada por ARL
  fecha_aprobacion_arl DATE,
  consultor TEXT,               -- email del consultor que la generó
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now()
);

-- PLAN DE EMERGENCIAS
CREATE TABLE planes_emergencia (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  empresa_id UUID REFERENCES empresas(id),
  version INTEGER DEFAULT 1,
  -- Audios por sección
  audio_amenazas_url TEXT,
  audio_recursos_url TEXT,
  audio_vulnerabilidades_url TEXT,
  audio_estructura_respuesta_url TEXT,
  audio_inventario_url TEXT,
  -- Transcripciones por sección
  transcripcion_amenazas TEXT,
  transcripcion_recursos TEXT,
  transcripcion_vulnerabilidades TEXT,
  transcripcion_estructura TEXT,
  transcripcion_inventario TEXT,
  -- Documento generado
  analisis_vulnerabilidad JSONB,
  documento_generado TEXT,
  drive_file_id TEXT,
  estado TEXT CHECK (estado IN ('en_proceso','borrador','revisado','aprobado')) DEFAULT 'en_proceso',
  created_at TIMESTAMPTZ DEFAULT now()
);

-- DOCUMENTOS DE CUMPLIMIENTO (motor de puntos Res. 0312)
CREATE TABLE documentos_cumplimiento (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  empresa_id UUID REFERENCES empresas(id),
  estandar TEXT,                -- Ej: 'I. Recursos'
  criterio_codigo TEXT,         -- Ej: '1.1.1'
  criterio_descripcion TEXT,
  peso_porcentual DECIMAL(5,2), -- peso según Res. 0312
  estado TEXT CHECK (estado IN ('cumple','no_cumple','no_aplica','en_proceso')) DEFAULT 'no_cumple',
  evidencia_descripcion TEXT,
  drive_file_id TEXT,           -- documento de evidencia
  fecha_carga TIMESTAMPTZ,
  consultor TEXT,
  observaciones TEXT,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now()
);

-- CIIU (tabla de referencia)
CREATE TABLE ciiu (
  codigo TEXT PRIMARY KEY,
  seccion TEXT,
  division TEXT,
  grupo TEXT,
  clase TEXT,
  descripcion TEXT,
  descripcion_detallada TEXT,
  -- Riesgos típicos pre-clasificados para Módulo 3
  riesgos_fisicos JSONB,
  riesgos_quimicos JSONB,
  riesgos_biologicos JSONB,
  riesgos_psicosociales JSONB,
  riesgos_ergonomicos JSONB,
  riesgos_locativos JSONB,
  riesgos_mecanicos JSONB,
  riesgos_electricos JSONB,
  nivel_riesgo_predominante TEXT
);

-- PARAMETROS DEL SISTEMA
CREATE TABLE parametros_sistema (
  clave TEXT PRIMARY KEY,
  valor TEXT,
  descripcion TEXT,
  updated_at TIMESTAMPTZ DEFAULT now()
);
```

## 5. SKILLS ADAPTADAS — PROMPTS COMPLETOS PARA CLAUDE API

### SKILL 1 — EXTRACTOR_MEDICO_SST

**Propósito**: Extraer recomendaciones y restricciones de PDFs de exámenes médicos ocupacionales.

```javascript
// Llamada Claude API — Módulo 2
const SYSTEM_EXTRACTOR_MEDICO = `
Eres un especialista en medicina laboral colombiana. Tu función exclusiva es extraer información estructurada de exámenes médicos ocupacionales emitidos bajo la normativa colombiana (Resolución 2346 de 2007 del MinProtección Social, Decreto 1477 de 2014).

REGLAS DE EXTRACCIÓN:
1. Extrae ÚNICAMENTE lo que está explícitamente en el documento. No inferir ni completar.
2. Si un campo no existe en el documento, devuelve null para ese campo.
3. Las restricciones y recomendaciones van en listas separadas, no mezcladas.
4. El concepto médico debe ser exactamente uno de: "apto", "apto_con_restricciones", "no_apto", "pendiente".
5. Si hay ambigüedad en el concepto, clasifica como "apto_con_restricciones" y registra la ambigüedad en observaciones.

RESPONDE ÚNICAMENTE EN JSON VÁLIDO. Sin explicaciones previas ni posteriores.

FORMATO DE RESPUESTA:
{
  "concepto": "apto|apto_con_restricciones|no_apto|pendiente",
  "medico_nombre": "string|null",
  "medico_tarjeta_profesional": "string|null",
  "fecha_examen": "YYYY-MM-DD|null",
  "tipo_examen": "ingreso|periodico|egreso|post_incapacidad|null",
  "restricciones": [
    {
      "tipo": "fisica|ambiental|horaria|otra",
      "descripcion": "descripción exacta de la restricción",
      "temporal": true|false,
      "duracion_dias": null|número
    }
  ],
  "recomendaciones": [
    {
      "categoria": "ergonomia|salud_mental|condicion_medica|pausas|otra",
      "descripcion": "descripción exacta de la recomendación"
    }
  ],
  "diagnosticos_relevantes": ["string"],
  "observaciones_adicionales": "string|null",
  "requiere_seguimiento": true|false,
  "fecha_proximo_examen_sugerida": "YYYY-MM-DD|null"
}`;

// Función de llamada
async function extraerExamenMedico(pdfBase64, mediaType = "application/pdf") {
  const response = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      model: "claude-sonnet-4-20250514",
      max_tokens: 1000,
```

## 11. CONFIGURACIÓN SUPABASE RLS — VISTA CLIENTE

```sql
-- Habilitar RLS en todas las tablas
ALTER TABLE empresas ENABLE ROW LEVEL SECURITY;
ALTER TABLE documentos_cumplimiento ENABLE ROW LEVEL SECURITY;
ALTER TABLE examenes_medicos ENABLE ROW LEVEL SECURITY;

-- Tabla de usuarios-empresa (relaciona auth.users con empresa)
CREATE TABLE usuario_empresa (
  user_id UUID REFERENCES auth.users(id),
  empresa_id UUID REFERENCES empresas(id),
  rol TEXT CHECK (rol IN ('admin','cliente','consultor')),
  PRIMARY KEY (user_id, empresa_id)
);

-- Policy: Admin ve todo
CREATE POLICY "admin_all" ON empresas
  FOR ALL USING (
    EXISTS (SELECT 1 FROM usuario_empresa 
    WHERE user_id = auth.uid() AND rol = 'admin')
  );

-- Policy: Cliente solo ve su empresa
CREATE POLICY "cliente_propia_empresa" ON empresas
  FOR SELECT USING (
    EXISTS (SELECT 1 FROM usuario_empresa 
    WHERE user_id = auth.uid() AND empresa_id = empresas.id)
  );

-- Aplicar misma lógica a documentos_cumplimiento
CREATE POLICY "cliente_propia_cumplimiento" ON documentos_cumplimiento
  FOR SELECT USING (
    EXISTS (SELECT 1 FROM usuario_empresa 
    WHERE user_id = auth.uid() AND empresa_id = documentos_cumplimiento.empresa_id)
  );
```
